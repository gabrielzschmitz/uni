\section{Exercícios da Lista 1}

A primeira lista de exercícios tem como objetivo revisar técnicas de integração
indefinida, fundamentais para o desenvolvimento dos conteúdos de Equações
Diferenciais Ordinárias. Os exercícios abrangem diferentes técnicas de
integração, incluindo substituição, integração por partes, identidades
trigonométricas e frações parciais.

\question{
  Resolva as seguintes integrais indefinidas:
}

\subquestion{
  \[
    \int -\cot{g(x)}\,\csc{x}\,dx
  \]
}

\answer{
  Aplicando a relação
  \[
    \frac{d}{dx}\csc{x}=-\csc{x}\cot{x},
  \]
  temos
  \[
    \begin{aligned}
      \int -\cot{x}\,\csc{x}\,dx
      &= \csc{x}+C.
    \end{aligned}
  \]
  Portanto,
  \[
    \boxed{\int -\cot{x}\,\csc{x}\,dx=\csc{x}+C}.
  \]
}

\subquestion{
  \[
    \int \sec^{2}{x}\,dx
  \]
}

\answer{
  Aplicando a relação
  \[
    \frac{d}{dx}\tan{x}=\sec^{2}{x},
  \]
  obtemos
  \[
    \boxed{\int\sec^{2}{x}\,dx=\tan{x}+C}.
  \]
}

\subquestion{
  \[
    \int \sec{(5x)}\tan{(5x)}\,dx
  \]
}

\answer{
  Aplicando a regra da cadeia, fazemos
  \[
    \begin{aligned}
      u &= 5x, \\
      du &= 5\,dx, \\
      dx &= \frac{du}{5}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\sec(5x)\tan(5x)\,dx
      &= \frac{1}{5}\int\sec(u)\tan(u)\,du \\
      &= \frac{1}{5}\sec(u)+C \\
      &= \boxed{\frac{1}{5}\sec(5x)+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \frac{1}{1+x^{2}}\,dx
  \]
}

\answer{
  Aplicando a integral imediata da derivada da função arco-tangente,
  \[
    \frac{d}{dx}\arctan{x}
    =
    \frac{1}{1+x^{2}},
  \]
  temos
  \[
    \boxed{\int\frac{1}{1+x^{2}}\,dx=\arctan{x}+C}.
  \]
}

\subquestion{
  \[
    \int \csc^{2}{x}\,dx
  \]
}

\answer{
  Aplicando a relação
  \[
    \frac{d}{dx}\cot{x}=-\csc^{2}{x},
  \]
  obtemos
  \[
    \boxed{\int\csc^{2}{x}\,dx=-\cot{x}+C}.
  \]
}

\subquestion{
  \[
    \int \frac{\ln{x}}{x\ln{x^{2}}}\,dx
  \]
}

\answer{
  Aplicando a propriedade dos logaritmos,
  \[
    \ln(x^{2})=2\ln{x},
  \]
  podemos simplificar a integral:
  \[
    \begin{aligned}
      \int\frac{\ln{x}}{x\ln{x^{2}}}\,dx
      &= \int\frac{\ln{x}}{2x\ln{x}}\,dx \\
      &= \frac{1}{2}\int\frac{1}{x}\,dx \\
      &= \boxed{\frac{1}{2}\ln|x|+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \frac{x^{2}-1}{x^{2}+1}\,dx
  \]
}

\answer{
  Aplicando divisão algébrica, temos
  \[
    \begin{aligned}
      \frac{x^{2}-1}{x^{2}+1}
      &= \frac{x^{2}+1-2}{x^{2}+1} \\
      &= 1-\frac{2}{x^{2}+1}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int\frac{x^{2}-1}{x^{2}+1}\,dx
      &= \int 1\,dx
      -2\int\frac{1}{x^{2}+1}\,dx \\
      &= x-2\arctan{x}+C.
    \end{aligned}
  \]
  Logo,
  \[
    \boxed{x-2\arctan{x}+C}.
  \]
}

\subquestion{
  \[
    \int \tan^{2}{x}\,\csc^{2}{x}\,dx
  \]
}

\answer{
  Aplicando as identidades trigonométricas,
  \[
    \begin{aligned}
      \tan^{2}{x}\,\csc^{2}{x}
      &= \frac{\sin^{2}{x}}{\cos^{2}{x}}
         \frac{1}{\sin^{2}{x}} \\
      &= \frac{1}{\cos^{2}{x}} \\
      &= \sec^{2}{x}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\tan^{2}{x}\,\csc^{2}{x}\,dx
      &= \int\sec^{2}{x}\,dx \\
      &= \boxed{\tan{x}+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int x^{2}\sqrt{1+x}\,dx
  \]
}

\answer{
  Aplicando substituição, fazemos
  \[
    \begin{aligned}
      u &= 1+x, \\
      x &= u-1, \\
      dx &= du.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int x^{2}\sqrt{1+x}\,dx
      &= \int (u-1)^{2}u^{1/2}\,du \\
      &= \int\left(u^{5/2}-2u^{3/2}+u^{1/2}\right)\,du \\
      &= \frac{2}{7}u^{7/2}
         -\frac{4}{5}u^{5/2}
         +\frac{2}{3}u^{3/2}+C.
    \end{aligned}
  \]
  Retornando para $x$,
  \[
    \boxed{
      \frac{2}{7}(1+x)^{7/2}
      -\frac{4}{5}(1+x)^{5/2}
      +\frac{2}{3}(1+x)^{3/2}+C
    }.
  \]
}

\subquestion{
  \[
    \int xe^{3x^{2}}\,dx
  \]
}

\answer{
  Aplicando substituição, fazemos
  \[
    \begin{aligned}
      u &= 3x^{2}, \\
      du &= 6x\,dx, \\
      x\,dx &= \frac{du}{6}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int xe^{3x^{2}}\,dx
      &= \frac{1}{6}\int e^{u}\,du \\
      &= \frac{1}{6}e^{u}+C \\
      &= \boxed{\frac{1}{6}e^{3x^{2}}+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int x^{2}\sin{x}\,dx
  \]
}

\answer{
  Aplicando integração por partes, fazemos
  \[
    \begin{aligned}
      u &= x^{2}, \\
      dv &= \sin{x}\,dx, \\
      du &= 2x\,dx, \\
      v &= -\cos{x}.
    \end{aligned}
  \]
  Logo,
  \[
    \begin{aligned}
      \int x^{2}\sin{x}\,dx
      &= -x^{2}\cos{x}
         +2\int x\cos{x}\,dx.
    \end{aligned}
  \]
  Aplicando integração por partes novamente,
  \[
    \begin{aligned}
      u &= x, \\
      dv &= \cos{x}\,dx, \\
      du &= dx, \\
      v &= \sin{x}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int x\cos{x}\,dx
      &= x\sin{x}-\int\sin{x}\,dx \\
      &= x\sin{x}+\cos{x}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int x^{2}\sin{x}\,dx
      &= -x^{2}\cos{x}
         +2x\sin{x}+2\cos{x}+C.
    \end{aligned}
  \]
  Portanto,
  \[
    \boxed{-x^{2}\cos{x}+2x\sin{x}+2\cos{x}+C}.
  \]
}

\subquestion{
  \[
    \int e^{x}\cos{x}\,dx
  \]
}

\answer{
  Aplicando integração por partes, fazemos
  \[
    \begin{aligned}
      u &= \cos{x}, \\
      dv &= e^{x}\,dx, \\
      du &= -\sin{x}\,dx, \\
      v &= e^{x}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int e^{x}\cos{x}\,dx
      &= e^{x}\cos{x}
         +\int e^{x}\sin{x}\,dx.
    \end{aligned}
  \]
  Aplicando integração por partes novamente,
  \[
    \begin{aligned}
      u &= \sin{x}, \\
      dv &= e^{x}\,dx, \\
      du &= \cos{x}\,dx, \\
      v &= e^{x}.
    \end{aligned}
  \]
  Então,
  \[
    \begin{aligned}
      \int e^{x}\sin{x}\,dx
      &= e^{x}\sin{x}
         -\int e^{x}\cos{x}\,dx.
    \end{aligned}
  \]
  Substituindo,
  \[
    \begin{aligned}
      \int e^{x}\cos{x}\,dx
      &= e^{x}\cos{x}
         +e^{x}\sin{x}
         -\int e^{x}\cos{x}\,dx.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      2\int e^{x}\cos{x}\,dx
      &= e^{x}(\cos{x}+\sin{x}),
    \end{aligned}
  \]
  e finalmente
  \[
    \boxed{
      \int e^{x}\cos{x}\,dx
      =
      \frac{e^{x}}{2}
      (\cos{x}+\sin{x})+C
    }.
  \]
}

\subquestion{
  \[
    \int x^{2}e^{x}\,dx
  \]
}

\answer{
  Aplicando integração por partes, fazemos
  \[
    \begin{aligned}
      u &= x^{2}, \\
      dv &= e^{x}\,dx, \\
      du &= 2x\,dx, \\
      v &= e^{x}.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int x^{2}e^{x}\,dx
      &= x^{2}e^{x}
         -2\int xe^{x}\,dx.
    \end{aligned}
  \]
  Aplicando integração por partes novamente,
  \[
    \begin{aligned}
      u &= x, \\
      dv &= e^{x}\,dx, \\
      du &= dx, \\
      v &= e^{x}.
    \end{aligned}
  \]
  Logo,
  \[
    \begin{aligned}
      \int xe^{x}\,dx
      &= xe^{x}-\int e^{x}\,dx \\
      &= xe^{x}-e^{x}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int x^{2}e^{x}\,dx
      &= x^{2}e^{x}-2xe^{x}+2e^{x} \\
      &= \boxed{e^{x}(x^{2}-2x+2)+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \frac{2\sec^{2}{\theta}}{a+b\tan{\theta}}\,d\theta
  \]
}

\answer{
  Aplicando substituição, fazemos
  \[
    \begin{aligned}
      u &= a+b\tan{\theta}, \\
      du &= b\sec^{2}{\theta}\,d\theta.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\frac{2\sec^{2}{\theta}}
      {a+b\tan{\theta}}\,d\theta
      &= \frac{2}{b}\int\frac{1}{u}\,du \\
      &= \frac{2}{b}\ln|u|+C \\
      &= \boxed{
        \frac{2}{b}\ln|a+b\tan{\theta}|+C
      }.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \sqrt{x}\ln{x}\,dx
  \]
}

\answer{
  Aplicando integração por partes, fazemos
  \[
    \begin{aligned}
      u &= \ln{x}, \\
      dv &= x^{1/2}\,dx, \\
      du &= \frac{1}{x}\,dx, \\
      v &= \frac{2}{3}x^{3/2}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int x^{1/2}\ln{x}\,dx
      &= \frac{2}{3}x^{3/2}\ln{x}
         -\frac{2}{3}\int x^{1/2}\,dx \\
      &= \frac{2}{3}x^{3/2}\ln{x}
         -\frac{4}{9}x^{3/2}+C.
    \end{aligned}
  \]
  Logo,
  \[
    \boxed{
      \frac{2}{3}x^{3/2}\ln{x}
      -\frac{4}{9}x^{3/2}+C
    }.
  \]
}

\subquestion{
  \[
    \int \ln{(1-x)}\,dx
  \]
}

\answer{
  Aplicando substituição, fazemos
  \[
    \begin{aligned}
      u &= 1-x, \\
      du &= -dx, \\
      dx &= -du.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\ln(1-x)\,dx
      &= -\int\ln{u}\,du.
    \end{aligned}
  \]
  Aplicando integração por partes,
  \[
    \begin{aligned}
      u_1 &= \ln{u}, \\
      dv_1 &= du, \\
      du_1 &= \frac{1}{u}\,du, \\
      v_1 &= u.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      -\int\ln{u}\,du
      &= -(u\ln{u}-u)+C \\
      &= -u\ln{u}+u+C.
    \end{aligned}
  \]
  Retornando para $x$,
  \[
    \boxed{
      (1-x)\left[1-\ln|1-x|\right]+C
    }.
  \]
}

\subquestion{
  \[
    \int \frac{\sin{x}}{\cos^{5}{x}}\,dx
  \]
}

\answer{
  Aplicando substituição, fazemos
  \[
    \begin{aligned}
      u &= \cos{x}, \\
      du &= -\sin{x}\,dx.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\frac{\sin{x}}{\cos^{5}{x}}\,dx
      &= -\int u^{-5}\,du \\
      &= -\frac{u^{-4}}{-4}+C \\
      &= \frac{1}{4}u^{-4}+C \\
      &= \boxed{\frac{1}{4\cos^{4}{x}}+C}.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \cos^{5}{x}\,dx
  \]
}

\answer{
  Aplicando a identidade
  \[
    \cos^{5}{x}
    =
    \cos{x}(1-\sin^{2}{x})^{2},
  \]
  fazemos a substituição
  \[
    \begin{aligned}
      u &= \sin{x}, \\
      du &= \cos{x}\,dx.
    \end{aligned}
  \]
  Então,
  \[
    \begin{aligned}
      \int\cos^{5}{x}\,dx
      &= \int(1-u^{2})^{2}\,du \\
      &= \int(1-2u^{2}+u^{4})\,du \\
      &= u-\frac{2}{3}u^{3}
         +\frac{1}{5}u^{5}+C.
    \end{aligned}
  \]
  Retornando para $x$,
  \[
    \boxed{
      \sin{x}
      -\frac{2}{3}\sin^{3}{x}
      +\frac{1}{5}\sin^{5}{x}+C
    }.
  \]
}

\subquestion{
  \[
    \int \cos^{4}{x}\,dx
  \]
}

\answer{
  Aplicando a identidade de redução de potência,
  \[
    \cos^{4}{x}
    =
    \frac{3}{8}
    +\frac{1}{2}\cos(2x)
    +\frac{1}{8}\cos(4x).
  \]
  Portanto,
  \[
    \begin{aligned}
      \int\cos^{4}{x}\,dx
      &= \frac{3}{8}\int dx
         +\frac{1}{2}\int\cos(2x)\,dx
         +\frac{1}{8}\int\cos(4x)\,dx \\
      &= \frac{3x}{8}
         +\frac{\sin(2x)}{4}
         +\frac{\sin(4x)}{32}+C.
    \end{aligned}
  \]
  Logo,
  \[
    \boxed{
      \frac{3x}{8}
      +\frac{\sin(2x)}{4}
      +\frac{\sin(4x)}{32}+C
    }.
  \]
}

\subquestion{
  \[
    \int \tan^{3}{x}\,dx
  \]
}

\answer{
  Aplicando a identidade
  \[
    \tan^{2}{x}=\sec^{2}{x}-1,
  \]
  temos
  \[
    \begin{aligned}
      \int\tan^{3}{x}\,dx
      &= \int\tan{x}(\sec^{2}{x}-1)\,dx \\
      &= \int\tan{x}\sec^{2}{x}\,dx
         -\int\tan{x}\,dx.
    \end{aligned}
  \]
  Aplicando substituição na primeira integral,
  \[
    \begin{aligned}
      u &= \tan{x}, \\
      du &= \sec^{2}{x}\,dx,
    \end{aligned}
  \]
  obtemos
  \[
    \begin{aligned}
      \int\tan{x}\sec^{2}{x}\,dx
      &= \frac{1}{2}\tan^{2}{x}.
    \end{aligned}
  \]
  Além disso,
  \[
    \int\tan{x}\,dx=-\ln|\cos{x}|.
  \]
  Portanto,
  \[
    \boxed{
      \frac{1}{2}\tan^{2}{x}
      +\ln|\cos{x}|+C
    }.
  \]
}

\subquestion{
  \[
    \int \frac{dx}{x^{2}-4}
  \]
}

\answer{
  Aplicando fatoração no denominador,
  \[
    x^{2}-4=(x-2)(x+2).
  \]
  Utilizando frações parciais, escrevemos
  \[
    \begin{aligned}
      \frac{1}{(x-2)(x+2)}
      &= \frac{A}{x-2}+\frac{B}{x+2}.
    \end{aligned}
  \]
  Multiplicando pelo denominador,
  \[
    \begin{aligned}
      1
      &= A(x+2)+B(x-2).
    \end{aligned}
  \]
  Comparando os coeficientes,
  \[
    \begin{aligned}
      A+B &= 0, \\
      2A-2B &= 1,
    \end{aligned}
  \]
  logo,
  \[
    \begin{aligned}
      A &= \frac{1}{4}, \\
      B &= -\frac{1}{4}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int\frac{dx}{x^{2}-4}
      &= \frac{1}{4}\int\frac{dx}{x-2}
         -\frac{1}{4}\int\frac{dx}{x+2} \\
      &= \boxed{
        \frac{1}{4}\ln|x-2|
        -\frac{1}{4}\ln|x+2|+C
      }.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int \frac{4x^{2}+13x-9}
    {x^{3}+2x^{2}-3x}\,dx
  \]
}

\answer{
  Aplicando fatoração no denominador,
  \[
    \begin{aligned}
      x^{3}+2x^{2}-3x
      &= x(x^{2}+2x-3) \\
      &= x(x+3)(x-1).
    \end{aligned}
  \]
  Utilizando frações parciais, escrevemos
  \[
    \begin{aligned}
      \frac{4x^{2}+13x-9}
      {x(x+3)(x-1)}
      &=
      \frac{A}{x}
      +\frac{B}{x+3}
      +\frac{D}{x-1}.
    \end{aligned}
  \]
  Resolvendo os coeficientes, obtemos
  \[
    \begin{aligned}
      A &= 3, \\
      B &= -1, \\
      D &= 2.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int\frac{4x^{2}+13x-9}
      {x^{3}+2x^{2}-3x}\,dx
      &= \int
      \left(
        \frac{3}{x}
        -\frac{1}{x+3}
        +\frac{2}{x-1}
      \right)\,dx \\
      &= 3\ln|x|
         -\ln|x+3|
         +2\ln|x-1|+C.
    \end{aligned}
  \]
  Portanto,
  \[
    \boxed{
      3\ln|x|
      -\ln|x+3|
      +2\ln|x-1|+C
    }.
  \]
}

\subquestion{
  \[
    \int \frac{x-2}
    {x^{3}-3x^{2}-x+3}\,dx
  \]
}

\answer{
  Aplicando fatoração por agrupamento no denominador,
  \[
    \begin{aligned}
      x^{3}-3x^{2}-x+3
      &= x^{2}(x-3)-(x-3) \\
      &= (x^{2}-1)(x-3) \\
      &= (x-1)(x+1)(x-3).
    \end{aligned}
  \]
  Utilizando frações parciais,
  \[
    \begin{aligned}
      \frac{x-2}
      {(x-1)(x+1)(x-3)}
      &=
      \frac{A}{x-1}
      +\frac{B}{x+1}
      +\frac{D}{x-3}.
    \end{aligned}
  \]
  Resolvendo os coeficientes,
  \[
    \begin{aligned}
      A &= \frac{1}{4}, \\
      B &= -\frac{3}{8}, \\
      D &= \frac{1}{8}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int\frac{x-2}
      {x^{3}-3x^{2}-x+3}\,dx
      &= \frac{1}{4}\int\frac{dx}{x-1}
         -\frac{3}{8}\int\frac{dx}{x+1}
         +\frac{1}{8}\int\frac{dx}{x-3} \\
      &= \boxed{
        \frac{1}{4}\ln|x-1|
        -\frac{3}{8}\ln|x+1|
        +\frac{1}{8}\ln|x-3|+C
      }.
    \end{aligned}
  \]
}

\subquestion{
  \[
    \int
    \frac{3x^{3}-18x^{2}+29x-4}
    {(x+1)(x-2)^{3}}\,dx
  \]
}

\answer{
  Aplicando frações parciais, escrevemos
  \[
    \begin{aligned}
      \frac{3x^{3}-18x^{2}+29x-4}
      {(x+1)(x-2)^{3}}
      &=
      \frac{A}{x+1}
      +\frac{B}{x-2}
      +\frac{D}{(x-2)^{2}}
      +\frac{E}{(x-2)^{3}}.
    \end{aligned}
  \]
  Resolvendo os coeficientes, obtemos
  \[
    \begin{aligned}
      A &= 2, \\
      B &= 1, \\
      D &= -3, \\
      E &= 2.
    \end{aligned}
  \]
  Assim,
  \[
    \begin{aligned}
      \int
      \frac{3x^{3}-18x^{2}+29x-4}
      {(x+1)(x-2)^{3}}\,dx
      &=
      \int\left(
        \frac{2}{x+1}
        +\frac{1}{x-2}
        -\frac{3}{(x-2)^{2}}
        +\frac{2}{(x-2)^{3}}
      \right)\,dx \\
      &= 2\ln|x+1|
         +\ln|x-2|
         +\frac{3}{x-2}
         -\frac{1}{(x-2)^{2}}+C.
    \end{aligned}
  \]
  Portanto,
  \[
    \boxed{
      2\ln|x+1|
      +\ln|x-2|
      +\frac{3}{x-2}
      -\frac{1}{(x-2)^{2}}+C
    }.
  \]
}

\subquestion{
  \[
    \int
    \frac{-4x^{3}}
    {2x^{3}+x^{2}-2x-1}\,dx
  \]
}

\answer{
  Aplicando fatoração por agrupamento no denominador,
  \[
    \begin{aligned}
      2x^{3}+x^{2}-2x-1
      &= x^{2}(2x+1)-(2x+1) \\
      &= (x^{2}-1)(2x+1) \\
      &= (x-1)(x+1)(2x+1).
    \end{aligned}
  \]
  Como o grau do numerador é igual ao grau do denominador,
  aplicamos inicialmente divisão polinomial:
  \[
    \begin{aligned}
      \frac{-4x^{3}}
      {2x^{3}+x^{2}-2x-1}
      &=
      -2
      +\frac{2x^{2}-4x-2}
      {(x-1)(x+1)(2x+1)}.
    \end{aligned}
  \]
  Utilizando frações parciais na parte restante,
  \[
    \begin{aligned}
      \frac{2x^{2}-4x-2}
      {(x-1)(x+1)(2x+1)}
      &=
      \frac{2}{x+1}
      -\frac{2}{3(x-1)}
      -\frac{2}{3(2x+1)}.
    \end{aligned}
  \]
  Portanto,
  \[
    \begin{aligned}
      \int
      \frac{-4x^{3}}
      {2x^{3}+x^{2}-2x-1}\,dx
      &= \int\left(
        -2
        +\frac{2}{x+1}
        -\frac{2}{3(x-1)}
        -\frac{2}{3(2x+1)}
      \right)\,dx \\
      &= -2x
         +2\ln|x+1|
         -\frac{2}{3}\ln|x-1|
         -\frac{1}{3}\ln|2x+1|
         +C.
    \end{aligned}
  \]
